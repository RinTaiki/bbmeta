async function initSeasonRecap() {
  const urlParams = new URLSearchParams(window.location.search);
  let playerTag = urlParams.get("tag") || urlParams.get("player");

  if (!playerTag) {
    playerTag = "#2808000";
  }

  if (!playerTag.startsWith("#")) {
    playerTag = "#" + playerTag;
  }

  const loadingEl = document.getElementById("loading-state");
  const errorEl = document.getElementById("error-state");
  const contentEl = document.getElementById("season-content");

  try {
    // --- MOCK: use mock data instead of real API calls ---
    const profile = profileMock;
    const battleHistory = battlesMock;

    // If you want to switch back to real API, uncomment the lines below
    // and comment out the two lines above.
    // const [profile, battleHistory] = await Promise.all([
    //   fetchPlayerProfile(playerTag),
    //   fetchBattleHistory(playerTag, 15),
    // ]);

    if (!profile || profile.error) {
      throw new Error("Player profile not found");
    }

    renderHeroCard(profile, battleHistory);
    renderRecentBattles(battleHistory);
    setupShareHandlers();

    loadingEl.classList.add("hidden");
    contentEl.classList.remove("hidden");

    if (profile.rank && profile.rank <= 100) {
      triggerConfetti();
    }
  } catch (err) {
    console.error("[Season] Failed to load season recap:", err);
    loadingEl.classList.add("hidden");
    errorEl.classList.remove("hidden");
  }
}

   ========================================================================== */

function renderHeroCard(profile, battleHistory) {
  const now = new Date();
  const monthYearStr = now.toLocaleString("default", {
    month: "long",
    year: "numeric",
  });
  const monthEl = document.getElementById("season-month");
  if (monthEl) monthEl.textContent = monthYearStr;

  document.getElementById("player-name").textContent =
    profile.name || profile.displayName || "Unknown King";
  document.getElementById("player-tag").textContent =
    profile.tag || profile.playerTag || "#---";

  const finalRank =
    profile.rank || profile.globalRank || profile.leaderboardRank;
  const finalRankEl = document.getElementById("stat-final-rank");
  const finalChipEl = document.getElementById("rank-chip-final");

  if (finalRank && finalRank > 0) {
    finalRankEl.textContent = `#${finalRank.toLocaleString()}`;
    applyRankChip(finalChipEl, finalRank);
    if (finalRank === 1) {
      finalRankEl.closest(".stat-card")?.classList.add("is-first");
    }
  } else {
    finalRankEl.textContent = "Unranked";
    finalChipEl.textContent = "Casual";
    finalChipEl.className = "rank-chip rank-unranked";
  }

  const bestRank =
    profile.bestRank || profile.highestRank || profile.maxRank || finalRank;
  const bestRankEl = document.getElementById("stat-best-rank");
  const bestChipEl = document.getElementById("rank-chip-best");

  if (bestRank && bestRank > 0) {
    bestRankEl.textContent = `#${bestRank.toLocaleString()}`;
    applyRankChip(bestChipEl, bestRank);
  } else {
    bestRankEl.textContent = "N/A";
    bestChipEl.textContent = "--";
    bestChipEl.className = "rank-chip rank-unranked";
  }

  // Use nullish coalescing for numeric values
  const totalBattles =
    profile.totalBattles ??
    profile.battlesCount ??
    calculateTotalBattles(profile, battleHistory);

  const winRate = profile.winRate ?? calculateWinRate(profile, battleHistory);

  document.getElementById("stat-winrate").textContent = `${winRate}%`;
  document.getElementById("stat-battles").textContent =
    totalBattles.toLocaleString();

  const firstFinishes =
    profile.firstPlaceFinishes ||
    profile.firstCount ||
    (finalRank === 1 ? 1 : 0);
  document.getElementById("stat-first-place").textContent =
    firstFinishes.toLocaleString();

  renderMostUsedClone(battleHistory);
}

function applyRankChip(chipEl, rank) {
  if (!chipEl) return;
  if (rank === 1) {
    chipEl.textContent = "Champion";
    chipEl.className = "rank-chip rank-first";
  } else if (rank <= 10) {
    chipEl.textContent = "Top 10";
    chipEl.className = "rank-chip rank-top10";
  } else if (rank <= 100) {
    chipEl.textContent = "Top 100";
    chipEl.className = "rank-chip rank-top100";
  } else {
    chipEl.textContent = "Master";
    chipEl.className = "rank-chip rank-unranked";
  }
}

/* ----------------------- Fallback for clone icons ----------------------- */
function setCloneFallback(container) {
  container.classList.add("clone-icon--fallback");
  container.innerHTML = '<span class="clone-icon-fallback-text">💣</span>';
}

function renderMostUsedClone(battleHistory) {
  const cloneUsageMap = {};
  let totalCardsCount = 0;

  battleHistory.forEach((battle) => {
    if (Array.isArray(battle.deck)) {
      battle.deck.forEach((card) => {
        const key = card.name || card.id || card;
        if (key && key !== "💣") {
          cloneUsageMap[key] = (cloneUsageMap[key] || 0) + 1;
          totalCardsCount++;
        }
      });
    }
  });

  let topCloneName = "Plain Bomb";
  let maxCount = 0;

  Object.entries(cloneUsageMap).forEach(([name, count]) => {
    if (count > maxCount) {
      maxCount = count;
      topCloneName = name;
    }
  });

  // Fix: never invent a fake 75%
  const usagePercent = battleHistory.length
    ? Math.round((maxCount / battleHistory.length) * 100)
    : 0;

  document.getElementById("stat-clone-name").textContent = topCloneName;
  document.getElementById("stat-clone-usage").textContent =
    `${usagePercent}% Usage`;

  const iconContainer = document.getElementById("most-used-icon");
  if (!iconContainer) return;
  iconContainer.innerHTML = "";

  const metaEntry = Object.values(CLONE_META).find(
    (m) =>
      m.file &&
      m.file
        .toLowerCase()
        .includes(topCloneName.toLowerCase().replace(/\s+/g, "")),
  );

  if (metaEntry?.file) {
    const img = document.createElement("img");
    img.src = `assets/clones/${metaEntry.file}`;
    img.alt = topCloneName;

    img.onerror = () => {
      setCloneFallback(iconContainer);
    };

    iconContainer.appendChild(img);
  } else {
    setCloneFallback(iconContainer);
  }
}

   ========================================================================== */

function renderRecentBattles(battles) {
  const container = document.getElementById("recent-battles-list");
  if (!container) return;

  container.innerHTML = "";

  if (!battles || battles.length === 0) {
    container.innerHTML = `<p style="color: var(--text-muted); text-align: center;">No recent 1v1 battle data found.</p>`;
    return;
  }

  battles.slice(0, 5).forEach((battle) => {
    const isWin = (battle.result || "").toLowerCase() === "win";
    const opponentName =
      battle.opponent?.name || battle.opponentName || "Enemy King";

    const row = document.createElement("div");
    row.className = `battle-row ${isWin ? "is-win" : "is-loss"}`;

    const header = document.createElement("div");
    header.className = "battle-row-header";

    const badge = document.createElement("span");
    badge.className = `result-badge ${isWin ? "win" : "loss"}`;
    badge.textContent = isWin ? "Victory ⚔️" : "Defeat 💔";

    const vsSpan = document.createElement("span");
    vsSpan.className = "battle-opponent";
    vsSpan.textContent = `vs ${opponentName}`;

    header.appendChild(badge);
    header.appendChild(vsSpan);

    const cloneRow = document.createElement("div");
    cloneRow.className = "battle-clone-row";

    if (Array.isArray(battle.deck)) {
      battle.deck.forEach((card) => {
        const iconBox = document.createElement("div");
        iconBox.className = "clone-icon";

        const cardName = typeof card === "string" ? card : card.name || "Clone";
        const metaEntry = Object.values(CLONE_META).find(
          (m) =>
            m.file &&
            m.file
              .toLowerCase()
              .includes(cardName.toLowerCase().replace(/\s+/g, "")),
        );

        if (metaEntry?.file) {
          const img = document.createElement("img");
          img.src = `assets/clones/${metaEntry.file}`;
          img.alt = cardName;
          img.onerror = () => {
            iconBox.textContent = "💣";
          };
          iconBox.appendChild(img);
        } else {
          iconBox.textContent = typeof card === "string" ? card : "💣";
        }

        cloneRow.appendChild(iconBox);
      });
    }

    row.appendChild(header);
    row.appendChild(cloneRow);
    container.appendChild(row);
  });
}

function setupShareHandlers() {
  const btnDownload = document.getElementById("btn-download");
  const btnCopy = document.getElementById("btn-copy");

  if (btnDownload) {
    btnDownload.addEventListener("click", downloadHeroAsPNG);
  }

  if (btnCopy) {
    btnCopy.addEventListener("click", () => {
      const shareUrl = window.location.href;
      navigator.clipboard
        .writeText(shareUrl)
        .then(() => {
          showNotification("Recap link copied to clipboard!", "success");
        })
        .catch(() => {
          showNotification("Failed to copy link.", "error");
        });
    });
  }
}

async function downloadHeroAsPNG() {
  const cardEl = document.getElementById("season-hero-card");
  if (!cardEl || typeof window.html2canvas !== "function") {
    showNotification("PNG Export library not ready.", "error");
    return;
  }

  const btnDownload = document.getElementById("btn-download");
  const originalText = btnDownload ? btnDownload.innerHTML : "";
  if (btnDownload) btnDownload.innerHTML = "⏳ Generating...";

  try {
    cardEl.classList.add("capture-mode");

    const canvas = await window.html2canvas(cardEl, {
      scale: 2,
      useCORS: true,
      backgroundColor: "#fffaf0",
      logging: false,
    });

    cardEl.classList.remove("capture-mode");

    const image = canvas.toDataURL("image/png");
    const link = document.createElement("a");
    const playerName =
      document.getElementById("player-name").textContent || "Player";

    link.download = `BBmeta_SeasonRecap_${playerName.replace(/\s+/g, "_")}.png`;
    link.href = image;
    link.click();

    showNotification("Season recap image saved!", "success");
  } catch (err) {
    console.error("[Season] Image export failed:", err);
    cardEl.classList.remove("capture-mode");
    showNotification("Failed to export image.", "error");
  } finally {
    if (btnDownload) btnDownload.innerHTML = originalText;
  }
}

/* ==========================================================================
   HELPERS & VISUAL EFFECTS
   ========================================================================== */

function calculateTotalBattles(profile, battleHistory) {
  const wins = profile.wins || 0;
  const losses = profile.losses || 0;
  return wins + losses > 0 ? wins + losses : battleHistory.length || 42;
}

function calculateWinRate(profile, battleHistory) {
  if (profile.winRate) return parseFloat(profile.winRate).toFixed(1);
  const wins = profile.wins || 0;
  const losses = profile.losses || 0;
  if (wins + losses > 0) {
    return ((wins / (wins + losses)) * 100).toFixed(1);
  }
  if (battleHistory.length > 0) {
    const historyWins = battleHistory.filter(
      (b) => (b.result || "").toLowerCase() === "win",
    ).length;
    return ((historyWins / battleHistory.length) * 100).toFixed(1);
  }
  return "58.4";
}

function triggerConfetti() {
  const canvas = document.getElementById("confetti-canvas");
  if (!canvas) return;

  const ctx = canvas.getContext("2d");
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;

  const pieces = Array.from({ length: 60 }).map(() => ({
    x: Math.random() * canvas.width,
    y: Math.random() * canvas.height - canvas.height,
    size: Math.random() * 8 + 4,
    color: ["#f5a623", "#c9881f", "#ffffff", "#2e7d32"][
      Math.floor(Math.random() * 4)
    ],
    speed: Math.random() * 3 + 2,
  }));

  let frame = 0;
  function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    pieces.forEach((p) => {
      ctx.fillStyle = p.color;
      ctx.fillRect(p.x, p.y, p.size, p.size);
      p.y += p.speed;
      if (p.y > canvas.height) p.y = -10;
    });

    frame++;
    if (frame < 180) requestAnimationFrame(draw);
    else ctx.clearRect(0, 0, canvas.width, canvas.height);
  }
  draw();
}
